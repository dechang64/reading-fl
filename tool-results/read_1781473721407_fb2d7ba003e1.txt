    97→            item['appear_start'] = 'no'
    98→        else:
    99→            item['appear_start'] = result

   394→def calculate_page_offset(pairs):
   395→    differences = []
   396→    for pair in pairs:
   397→        try:
   398→            physical_index = pair['physical_index']
   399→            page_number = pair['page']
   400→            difference = physical_index - page_number
   401→            differences.append(difference)
   402→        except (KeyError, TypeError):
   403→            continue
   404→    
   405→    if not differences:
   406→        return None
   407→    
   408→    difference_counts = {}
   409→    for diff in differences:
   410→        difference_counts[diff] = difference_counts.get(diff, 0) + 1
   411→    
   412→    most_common = max(difference_counts.items(), key=lambda x: x[1])[0]
   413→

   426→def page_list_to_group_text(page_contents, token_lengths, max_tokens=20000, overlap_page=1):    
   427→    num_tokens = sum(token_lengths)
   428→    
   429→    if num_tokens <= max_tokens:
   430→        # merge all pages into one text
   431→        page_text = "".join(page_contents)
   432→        return [page_text]
   433→    
   434→    subsets = []
   435→    current_subset = []
   436→    current_token_count = 0
   437→
   438→    expected_parts_num = math.ceil(num_tokens / max_tokens)
   439→    average_tokens_per_part = math.ceil(((num_tokens / expected_parts_num) + max_tokens) / 2)
   440→    
   441→    for i, (page_content, page_tokens) in enumerate(zip(page_contents, token_lengths)):
   442→        if current_token_count + page_tokens > average_tokens_per_part:
   443→
   444→            subsets.append(''.join(current_subset))
   445→            # Start new subset from overlap if specified
   446→            overlap_start = max(i - overlap_page, 0)
   447→            current_subset = page_contents[overlap_start:i]
   448→            current_token_count = sum(token_lengths[overlap_start:i])
   449→        
   450→        # Add current page to the subset
   451→        current_subset.append(page_content)
   452→        current_token_count += page_tokens
   453→
   454→    # Add the last subset if it contains any pages
   455→    if current_subset:
   456→        subsets.append(''.join(current_subset))
   457→    
   458→    print('divide page_list to groups', len(subsets))
   459→    return subsets

   760→async def fix_incorrect_toc(toc_with_page_number, page_list, incorrect_results, start_index=1, model=None, logger=None):
   761→    print(f'start fix_incorrect_toc with {len(incorrect_results)} incorrect results')
   762→    incorrect_indices = {result['list_index'] for result in incorrect_results}
   763→    
   764→    end_index = len(page_list) + start_index - 1
   765→    
   766→    incorrect_results_and_range_logs = []
   767→    # Helper function to process and check a single incorrect item
   768→    async def process_and_check_item(incorrect_item):
   769→        list_index = incorrect_item['list_index']
   770→        
   771→        # Check if list_index is valid
   772→        if list_index < 0 or list_index >= len(toc_with_page_number):
   773→            # Return an invalid result for out-of-bounds indices
   774→            return {
   775→                'list_index': list_index,
   776→                'title': incorrect_item['title'],
   777→                'physical_index': incorrect_item.get('physical_index'),
   778→                'is_valid': False
   779→            }
   780→        
   781→        # Find the previous correct item
   782→        prev_correct = None
   783→        for i in range(list_index-1, -1, -1):
   784→            if i not in incorrect_indices and i >= 0 and i < len(toc_with_page_number):
   785→                physical_index = toc_with_page_number[i].get('physical_index')
   786→                if physical_index is not None:
   787→                    prev_correct = physical_index
   788→                    break
   789→        # If no previous correct item found, use start_index
   790→        if prev_correct is None:
   791→            prev_correct = start_index - 1
   792→        
   793→        # Find the next correct item
   794→        next_correct = None
   795→        for i in range(list_index+1, len(toc_with_page_number)):
   796→            if i not in incorrect_indices and i >= 0 and i < len(toc_with_page_number):
   797→                physical_index = toc_with_page_number[i].get('physical_index')
   798→                if physical_index is not None:
   799→                    next_correct = physical_index
   800→                    break
   801→        # If no next correct item found, use end_index
   802→        if next_correct is None:
   803→            next_correct = end_index
   804→        
   805→        incorrect_results_and_range_logs.append({
   806→            'list_index': list_index,
   807→            'title': incorrect_item['title'],
   808→            'prev_correct': prev_correct,
   809→            'next_correct': next_correct
   810→        })
   811→
   812→        page_contents=[]
   813→        for page_index in range(prev_correct, next_correct+1):
   814→            # Add bounds checking to prevent IndexError
   815→            page_list_idx = page_index - start_index
   816→            if page_list_idx >= 0 and page_list_idx < len(page_list):
   817→                page_text = f"<physical_index_{page_index}>\n{page_list[page_list_idx][0]}\n<physical_index_{page_index}>\n\n"
   818→                page_contents.append(page_text)
   819→            else:
   820→                continue
   821→        content_range = ''.join(page_contents)
   822→        
   823→        physical_index_int = await single_toc_item_index_fixer(incorrect_item['title'], content_range, model)
   824→        
   825→        # Check if the result is correct
   826→        check_item = incorrect_item.copy()
   827→        check_item['physical_index'] = physical_index_int
   828→        check_result = await check_title_appearance(check_item, page_list, start_index, model)
   829→
   830→        return {
   831→            'list_index': list_index,
   832→            'title': incorrect_item['title'],
   833→            'physical_index': physical_index_int,
   834→            'is_valid': check_result['answer'] == 'yes'
   835→        }
   836→
   837→    # Process incorrect items concurrently
   838→    tasks = [
   839→        process_and_check_item(item)
   840→        for item in incorrect_results
   841→    ]
   842→    results = await asyncio.gather(*tasks, return_exceptions=True)
   843→    for item, result in zip(incorrect_results, results):
   844→        if isinstance(result, Exception):
   845→            print(f"Processing item {item} generated an exception: {result}")
   846→            continue

   848→
   849→    # Update the toc_with_page_number with the fixed indices and check for any invalid results
   850→    invalid_results = []
   851→    for result in results:
   852→        if result['is_valid']:
   853→            # Add bounds checking to prevent IndexError
   854→            list_idx = result['list_index']
   855→            if 0 <= list_idx < len(toc_with_page_number):
   856→                toc_with_page_number[list_idx]['physical_index'] = result['physical_index']
   857→            else:
   858→                # Index is out of bounds, treat as invalid
   859→                invalid_results.append({
   860→                    'list_index': result['list_index'],
   861→                    'title': result['title'],
   862→                    'physical_index': result['physical_index'],
   863→                })
   864→        else:
   865→            invalid_results.append({
   866→                'list_index': result['list_index'],
   867→                'title': result['title'],
   868→                'physical_index': result['physical_index'],
   869→            })
   870→
   871→    logger.info(f'incorrect_results_and_range_logs: {incorrect_results_and_range_logs}')

   874→    return toc_with_page_number, invalid_results

   901→    print('start verify_toc')

   907→            break
   908→    
   909→    # Early return if we don't have valid physical indices

   911→        return 0, []

   913→    # Determine which items to check

   915→        print('check all items')

   922→    # Prepare items with their list indices

   924→    for idx in sample_indices:
   925→        item = list_result[idx]

   934→        check_title_appearance(item, page_list, start_index, model)

   940→    correct_count = 0

  1124→def validate_and_truncate_physical_indices(toc_with_page_number, page_list_length, start_index=1, logger=None):
  1125→    """
  1126→    Validates and truncates physical indices that exceed the actual document length.
  1127→    This prevents errors when TOC references pages that don't exist in the document (e.g. the file is broken or incomplete).
  1128→    """
  1129→    if not toc_with_page_number:
  1130→        return toc_with_page_number
  1131→    
  1132→    max_allowed_page = page_list_length + start_index - 1
  1133→    truncated_items = []
  1134→    
  1135→    for i, item in enumerate(toc_with_page_number):
  1136→        if item.get('physical_index') is not None:
  1137→            original_index = item['physical_index']
  1138→            if original_index > max_allowed_page:
  1139→                item['physical_index'] = None
  1140→                truncated_items.append({
  1141→                    'title': item.get('title', 'Unknown'),
  1142→                    'original_index': original_index
  1143→                })
  1144→                if logger:
  1145→                    logger.info(f"Removed physical_index for '{item.get('title', 'Unknown')}' (was {original_index}, too far beyond document)")
  1146→    
  1147→    if truncated_items and logger:
  1148→        logger.info(f"Total removed items: {len(truncated_items)}")
  1149→        
  1150→    print(f"Document validation: {page_list_length} pages, max allowed index: {max_allowed_page}")
  1151→    if truncated_items:
  1152→        print(f"Truncated {len(truncated_items)} TOC items that exceeded document length")
  1153→     
  1154→    return toc_with_page_number
