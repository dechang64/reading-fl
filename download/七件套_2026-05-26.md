### 二、OPC Agent编排引擎

```python
"""
OPC Agent Orchestrator
一人公司的AI Agent编排引擎
"""
from dataclasses import dataclass, field
from typing import List, Dict, Optional, Callable
from enum import Enum
import asyncio

class AgentRole(Enum):
    """Agent角色定义"""
    DESIGNER = "设计师"      # 视觉生成
    WRITER = "写手"          # 文本生成
    ANALYST = "分析师"       # 数据分析
    REVIEWER = "审核员"      # 质量把控
    COORDINATOR = "协调员"   # 任务调度

@dataclass
class Task:
    """任务定义"""
    task_id: str
    description: str
    role: AgentRole
    dependencies: List[str] = field(default_factory=list)
    status: str = "pending"
    result: Optional[Dict] = None

class OPCOrchestrator:
    """OPC Agent编排引擎——一人公司的AI团队调度器"""

    def __init__(self, llm_client):
        self.llm = llm_client
        self.agents: Dict[AgentRole, Callable] = {}
        self.task_queue: List[Task] = []
        self.completed: Dict[str, Task] = {}

    def register_agent(self, role: AgentRole, handler: Callable):
        """注册Agent"""
        self.agents[role] = handler

    def submit_intent(self, intent: str) -> List[Task]:
        """
        提交创作意图，自动分解为任务链
        这是"深度协同"的核心：人出意图，AI分解执行
        """
        # 用LLM将意图分解为子任务
        decomposition = self.llm.generate(f"""
        将以下创作意图分解为具体的子任务，每个子任务指定角色：
        意图：{intent}

        可用角色：{[r.value for r in AgentRole]}

        输出JSON格式：
        [{{"description": "...", "role": "...", "dependencies": []}}]
        """)

        tasks = self._parse_tasks(decomposition)
        self.task_queue.extend(tasks)
        return tasks

    async def execute(self) -> Dict[str, Dict]:
        """执行任务链——按依赖关系调度Agent"""
        results = {}
        max_iterations = 10  # 防止死循环

        for _ in range(max_iterations):
            # 找到所有依赖已满足的待执行任务
            ready_tasks = [
                t for t in self.task_queue
                if t.status == "pending"
                and all(dep in self.completed for dep in t.dependencies)
            ]

            if not ready_tasks:
                break

            # 并行执行就绪任务
            coroutines = []
            for task in ready_tasks:
                task.status = "running"
                handler = self.agents.get(task.role)
                if handler:
                    coroutines.append(self._run_task(task, handler))

            task_results = await asyncio.gather(*coroutines, return_exceptions=True)

            for task, result in zip(ready_tasks, task_results):
                if isinstance(result, Exception):
                    task.status = "failed"
                    task.result = {"error": str(result)}
                else:
                    task.status = "completed"
                    task.result = result
                self.completed[task.task_id] = task
                results[task.task_id] = task.result

        return results

    async def _run_task(self, task: Task, handler: Callable) -> Dict:
        """执行单个任务"""
        # 将已完成依赖的结果注入上下文
        context = {
            dep_id: self.completed[dep_id].result
            for dep_id in task.dependencies
        }
        return await handler(task.description, context)

    def _parse_tasks(self, decomposition: str) -> List[Task]:
        """解析LLM输出的任务分解"""
        import json
        try:
            parsed = json.loads(decomposition)
            return [
                Task(
                    task_id=f"task_{i}",
                    description=t["description"],
                    role=AgentRole(t["role"]),
                    dependencies=t.get("dependencies", [])
                )
                for i, t in enumerate(parsed)
            ]
        except (json.JSONDecodeError, KeyError, ValueError):
            # 解析失败时返回默认任务
            return [Task(
                task_id="task_0",
                description=decomposition,
                role=AgentRole.COORDINATOR
            )]


# 使用示例
async def demo():
    """一人公司Agent编排演示"""
    # 模拟LLM客户端
    class MockLLM:
        def generate(self, prompt):
            return '[{"description": "设计封面", "role": "设计师", "dependencies": []}, {"description": "撰写文案", "role": "写手", "dependencies": []}, {"description": "审核整体质量", "role": "审核员", "dependencies": ["task_0", "task_1"]}]'

    orchestrator = OPCOrchestrator(MockLLM())
```

### 三、联邦学习：一人公司的数据主权

```python
"""
OPC-FL: 一人公司联邦学习框架
FedAvg加权聚合 + 数据主权保护
"""
import copy
import numpy as np
from typing import List, Dict

class FedAvgAggregator:
    """FedAvg加权聚合器——标准FedAvg公式"""

    @staticmethod
    def aggregate(client_params_list: List[Dict[str, np.ndarray]],
                  client_data_sizes: List[int]) -> Dict[str, np.ndarray]:
        """
        FedAvg加权聚合：w_global = Σ(n_k/N) * w_k
        Args:
            client_params_list: 各一人公司客户端的模型参数
            client_data_sizes: 各一人公司的数据量
        Returns:
            聚合后的全局模型参数
        """
        if not client_params_list:
            raise ValueError("No client parameters to aggregate")

        total_samples = sum(client_data_sizes)
        if total_samples == 0:
            raise ValueError("Total client data size is zero")

        aggregated = {}
        for key in client_params_list[0].keys():
            weighted_sum = np.zeros_like(client_params_list[0][key])
            for params, size in zip(client_params_list, client_data_sizes):
                weight = size / total_samples
                weighted_sum += weight * params[key]
            aggregated[key] = weighted_sum

        return aggregated


# 使用示例
```

### 一个人怎么开公司？

小杨不是这么干的。

"我想做一个帮小商家做社交媒体内容的服务，目标客户是餐饮店，预算控制在每月2000以内。"
